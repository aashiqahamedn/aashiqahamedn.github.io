import os import re import time import asyncio from urllib.parse import urlparse from playwright.async_api import async_playwright URL_FILE = "urls.txt" OUTPUT_DIR = "screenshots" CHROMIUM_PATH = "/usr/bin/chromium" def sanitize(text): return re.sub(r'[<>:"/\\|?*]', '_', text) async def capture(browser, url): parsed = urlparse(url) host = parsed.hostname or "unknown" port = parsed.port if not port: port = 443 if parsed.scheme == "https" else 80 folder = os.path.join( OUTPUT_DIR, str(port) ) os.makedirs(folder, exist_ok=True) filename = sanitize( f"{parsed.scheme}_{host}_{port}.png" ) outfile = os.path.join( folder, filename ) print(f"[START] {url}") context = await browser.new_context( ignore_https_errors=True, viewport={ "width": 1920, "height": 1080 } ) page = await context.new_page() try: try: await page.goto( url, timeout=15000, wait_until="load" ) except Exception as e: print(f"[INFO] Navigation Error: {url}") print(f" {e}") # Let Chromium render either page or error page await page.wait_for_timeout(5000) # Add URL banner try: await page.evaluate( """ (url) => { let div = document.createElement('div'); div.innerText = 'URL: ' + url; div.style.position = 'fixed'; div.style.top = '0'; div.style.left = '0'; div.style.width = '100%'; div.style.background = 'red'; div.style.color = 'white'; div.style.padding = '10px'; div.style.fontSize = '20px'; div.style.fontWeight = 'bold'; div.style.zIndex = '999999'; document.body.prepend(div); } """, url ) except: pass await page.screenshot( path=outfile, full_page=True ) print(f"[SAVED] {outfile}") except Exception as e: print(f"[FAILED] {url}") print(e) finally: await context.close() async def main(): if not os.path.exists(URL_FILE): print("urls.txt not found") return with open(URL_FILE, "r") as f: urls = [ x.strip() for x in f if x.strip() ] print(f"Loaded {len(urls)} URLs") os.makedirs( OUTPUT_DIR, exist_ok=True ) async with async_playwright() as p: browser = await p.chromium.launch( executable_path=CHROMIUM_PATH, headless=True, args=[ "--ignore-certificate-errors", "--no-sandbox", "--disable-dev-shm-usage" ] ) for url in urls: await capture(browser, url) await browser.close() if __name__ == "__main__": asyncio.run(main())
