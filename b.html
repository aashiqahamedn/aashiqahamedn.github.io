from selenium import webdriver from selenium.webdriver.firefox.options import Options import time opts = Options() opts.accept_insecure_certs = True driver = webdriver.Firefox(options=opts) driver.set_window_size(1920, 1080) url = "http://10.13.139.1:80" try: driver.get(url) except Exception as e: print("Navigation exception:", e) time.sleep(5) print("Current URL:", driver.current_url) print("Title:", driver.title) driver.save_screenshot("test.png") driver.quit()
