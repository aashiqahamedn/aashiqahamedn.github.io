import asyncio import os import re from urllib.parse import urlparse from PIL import Image from PIL import ImageDraw from PIL import ImageFont from playwright.async_api import async_playwright # CHANGE THIS PROFILE_DIR = "/home/kali/.config/chromium" CHROMIUM_PATH = "/usr/bin/chromium" URL_FILE = "urls.txt" OUTPUT_DIR = "screenshots" def sanitize(text): return re.sub(r'[<>:"/\\|?*]', "_", text) def add_banner(image_file, original_url, final_url): img = Image.open(image_file) banner_height = 100 canvas = Image.new( "RGB", (img.width, img.height + banner_height), "white" ) draw = ImageDraw.Draw(canvas) draw.rectangle( [(0, 0), (img.width, banner_height)], fill=(180, 0, 0) ) try: font = ImageFont.truetype( "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", 20 ) except: font = ImageFont.load_default() draw.text( (10, 10), f"Original: {original_url}", fill="white", font=font ) draw.text( (10, 50), f"Final: {final_url}", fill="white", font=font ) canvas.paste( img, (0, banner_height) ) canvas.save(image_file) async def capture(context, url): parsed = urlparse(url) host = parsed.hostname or "unknown" port = parsed.port if not port: port = 443 if parsed.scheme == "https" else 80 folder = os.path.join( OUTPUT_DIR, str(port) ) os.makedirs( folder, exist_ok=True ) filename = sanitize( f"{parsed.scheme}_{host}_{port}.png" ) outfile = os.path.join( folder, filename ) page = await context.new_page() final_url = url try: print(f"[START] {url}") try: await page.goto( url, timeout=30000, wait_until="networkidle" ) await page.wait_for_timeout( 5000 ) final_url = page.url except Exception as e: print( f"[INFO] Navigation Error: {url}" ) print(e) try: final_url = page.url except: pass await page.screenshot( path=outfile, full_page=True ) add_banner( outfile, url, final_url ) print( f"[SAVED] {outfile}" ) print( f"[FINAL] {final_url}" ) except Exception as e: print( f"[FAILED] {url}" ) print(e) finally: await page.close() async def main(): if not os.path.exists( URL_FILE ): print( f"{URL_FILE} not found" ) return with open( URL_FILE, "r" ) as f: urls = [ x.strip() for x in f if x.strip() ] print( f"Loaded {len(urls)} URLs" ) os.makedirs( OUTPUT_DIR, exist_ok=True ) async with async_playwright() as p: context = await p.chromium.launch_persistent_context( user_data_dir=PROFILE_DIR, executable_path=CHROMIUM_PATH, headless=False, ignore_https_errors=True, args=[ "--ignore-certificate-errors", "--disable-blink-features=AutomationControlled", "--start-maximized" ] ) for url in urls: await capture( context, url ) await context.close() if __name__ == "__main__": asyncio.run(main())
