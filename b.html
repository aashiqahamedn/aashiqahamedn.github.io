import os import re import time from urllib.parse import urlparse from PIL import Image from PIL import ImageDraw from PIL import ImageFont from selenium import webdriver from selenium.webdriver.firefox.options import Options from selenium.webdriver.firefox.service import Service URL_FILE = "urls.txt" OUTPUT_DIR = "screenshots" GECKODRIVER = "/usr/local/bin/geckodriver" def sanitize(text): return re.sub(r'[<>:"/\\|?*]', '_', text) def add_banner(image_file, original_url, final_url): img = Image.open(image_file) banner_height = 90 new_img = Image.new( "RGB", (img.width, img.height + banner_height), "white" ) draw = ImageDraw.Draw(new_img) draw.rectangle( [(0, 0), (img.width, banner_height)], fill=(180, 0, 0) ) try: font = ImageFont.truetype( "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", 20 ) except: font = ImageFont.load_default() draw.text( (10, 10), f"Original: {original_url}", fill="white", font=font ) draw.text( (10, 45), f"Final: {final_url}", fill="white", font=font ) new_img.paste(img, (0, banner_height)) new_img.save(image_file) def capture(url): parsed = urlparse(url) host = parsed.hostname or "unknown" port = parsed.port if not port: port = 443 if parsed.scheme == "https" else 80 folder = os.path.join( OUTPUT_DIR, str(port) ) os.makedirs(folder, exist_ok=True) outfile = os.path.join( folder, sanitize( f"{parsed.scheme}_{host}_{port}.png" ) ) opts = Options() opts.accept_insecure_certs = True service = Service( executable_path=GECKODRIVER ) driver = None try: driver = webdriver.Firefox( service=service, options=opts ) driver.set_window_size( 1920, 1080 ) try: driver.get(url) except Exception: pass time.sleep(8) final_url = driver.current_url print("=" * 80) print("Original :", url) print("Final :", final_url) print("Title :", driver.title) print("=" * 80) driver.save_screenshot(outfile) add_banner( outfile, url, final_url ) print(f"[SAVED] {outfile}") except Exception as e: print(f"[FAILED] {url}") print(e) finally: if driver: driver.quit() def main(): if not os.path.exists(URL_FILE): print("urls.txt not found") return with open(URL_FILE) as f: urls = [ x.strip() for x in f if x.strip() ] for url in urls: capture(url) if __name__ == "__main__": main()
