import os import time import subprocess from urllib.parse import urlparse URL_FILE = "urls.txt" OUTPUT_DIR = "screenshots" os.makedirs(OUTPUT_DIR, exist_ok=True) with open(URL_FILE) as f: urls = [u.strip() for u in f if u.strip()] for url in urls: parsed = urlparse(url) port = parsed.port if not port: port = 443 if parsed.scheme == "https" else 80 folder = os.path.join(OUTPUT_DIR, str(port)) os.makedirs(folder, exist_ok=True) filename = f"{parsed.scheme}_{parsed.hostname}_{port}.png" outfile = os.path.join(folder, filename) print(f"[+] Opening {url}") subprocess.Popen( ["chromium", url], stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL ) # Wait for page load / redirects time.sleep(10) # Bring Chromium to front subprocess.run( ["wmctrl", "-a", "Chromium"], stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL ) time.sleep(2) # Full screen screenshot subprocess.run( ["scrot", outfile] ) print(f"[+] Saved {outfile}") # Close current tab subprocess.run( ["xdotool", "key", "ctrl+w"], stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL ) time.sleep(2) print("Done")
